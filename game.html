// トークン取得 const token = localStorage.getItem("github_token"); if (!token) { alert("トークンが設定されていません。\ntoken.html で設定してください。"); } // GitHub API でファイルの存在だけ確認し、 // 実際の画像は raw.githubusercontent.com から取得する async function fetchImageUrl(path) { const apiUrl = `https://api.github.com/repos/maccccchon-collab/wordgame-assets/contents/${path}`; const response = await fetch(apiUrl, { headers: { Authorization: `token ${token}` } }); if (!response.ok) { console.error("画像を取得できませんでした:", path, response.status); return null; } // raw.githubusercontent.com の URL を自分で作る（これが最も確実） const rawUrl = `https://raw.githubusercontent.com/maccccchon-collab/wordgame-assets/main/${path}`; return rawUrl; } async function startGame() { const path = "images/1.jpg"; // ← 強制的に 1.jpg を表示 const imgUrl = await fetchImageUrl(path); console.log("1.jpg の取得結果:", imgUrl ? "成功" : "失敗"); if (imgUrl) { const img = document.createElement("img"); img.src = imgUrl; img.style.width = "200px"; document.body.appendChild(img); } else { const p = document.createElement("p"); p.textContent = "1.jpg を読み込めませんでした（404 の可能性）"; document.body.appendChild(p); } } startGame();
